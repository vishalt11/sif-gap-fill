\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.95\textwidth]{%
        \detokenize{img/Validation_channel_permutation_importance.pdf}}

    \caption[Validation permutation importance of predictor channels]{%
        Validation-set permutation importance of the predictor channels.
        Importance is measured as the increase in validation RMSE after
        independently permuting each predictor. The markers indicate the mean
        increase in RMSE, while the horizontal error bars represent variability
        across three permutation repeats. Larger positive values indicate
        greater predictive importance, whereas values close to or below zero
        indicate little contribution to validation performance. FAPAR, the
        seasonal cosine term, and NIRv show the greatest importance.}
    \label{fig:validation_channel_permutation_importance}
\end{figure}

        height=0.5\textheight,

    ]{\detokenize{img/test_spatial_residual_map_germany.pdf}}

    \caption[Spatial distribution of test residuals across Germany]{%
        Spatial distribution of the calibrated 4~km test residuals across
        Germany ($N=931$). Each grid cell is coloured according to the
        difference between predicted and observed SIF. Red cells indicate
        positive residuals and therefore model overestimation, whereas blue
        cells indicate negative residuals and model underestimation. Values
        close to white represent predictions close to the observations. The
        diverging colour scale is symmetrically clipped at the 98th percentile
        ($\pm 0.180$) to limit the visual influence of extreme residuals.}
    \label{fig:test_spatial_residual_map_germany}
\end{figure}
